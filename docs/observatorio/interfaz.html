150    def _actualizar_tabla(self) -> None:
151        """Actualiza el contenido visual de la tabla y la etiqueta de capacidad."""
152        for fila in self.tabla.get_children():
153            self.tabla.delete(fila)
154
155        for obj in self.catalogo.objetos:
156            ar, dec = obj.coordenadas
157            self.tabla.insert(
158                "", tk.END, values=(obj.identificador, obj.tipo, f"{ar:.2f}", f"{dec:.2f}")
159            )
160
161        self.lbl_estado.config(
162            text=f"Catálogo: {len(self.catalogo.objetos)} / {self.catalogo.capacidad_maxima} objetos"
163        )

165    def _on_anadir_objeto(self) -> None:
166        """Controlador para añadir un objeto recogiendo los datos del formulario."""
167        ident = self.txt_id.get().strip()
168        tipo_astro = self.txt_tipo.get().strip()
169        ar_str = self.txt_ar.get().strip()
170        dec_str = self.txt_dec.get().strip()
171
172        if not ident or not tipo_astro or not ar_str or not dec_str:
173            messagebox.showerror(
174                "Error de validación", "Todos los campos del formulario son obligatorios."
175            )
176            return
177
178        try:
179            ar = float(ar_str)
180            dec = float(dec_str)
181            nuevo_obj = ObjetoCeleste(ident, tipo_astro, (ar, dec))
182            self.catalogo.anadir(nuevo_obj)
183            self._actualizar_tabla()
184            messagebox.showinfo("Éxito", f"Objeto '{ident}' añadido correctamente al catálogo.")
185            self.txt_id.delete(0, tk.END)
186            self.txt_tipo.delete(0, tk.END)
187            self.txt_ar.delete(0, tk.END)
188            self.txt_dec.delete(0, tk.END)
189        except ValueError as e:
190            messagebox.showerror("Error de Formato / Rango", str(e))
191        except CapacidadMaximaError as e:
192            messagebox.showwarning("Capacidad Máxima", str(e))
193
194    def _on_crear_telescopio(self) -> None:
195        """Demuestra el uso del patrón Factory Method mediante un diálogo gráfico."""
196        tipo_optico = messagebox.askquestion(
197            "Patrón Factory",
198            "¿Qué tipo de telescopio deseas crear?\n\n- Sí: Refractor (Lentes)\n- No: Reflector (Espejos)",
199            icon=messagebox.QUESTION,
200        )
201        tipo_str = "refractor" if tipo_optico == "yes" else "reflector"
202        telescopio = FabricaTelescopios.crear_telescopio(tipo_str, "Telescopio GUI")
203        messagebox.showinfo(
204            "Telescopio Creado",
205            f"Instanciado mediante Factory:\n\n{telescopio.descripcion_tecnica()}",
206        )
207
208    def _on_guardar_json(self) -> None:
209        """Guarda el catálogo actual en fichero JSON usando las funciones de persistencia."""
210        try:
211            repo: Repositorio[ObjetoCeleste] = Repositorio()
212            for obj in self.catalogo.objetos:
213                repo.agregar(obj)
214            guardar_repositorio_json(repo, self.ruta_json, self.logger)
215            messagebox.showinfo(
216                "Persistencia", f"Catálogo guardado con éxito en '{self.ruta_json}'."
217            )
218        except PersistenciaError as e:
219            messagebox.showerror("Error de Persistencia", str(e))
220
221    def _on_cargar_json(self) -> None:
222        """Carga y reemplaza el catálogo desde un fichero JSON."""
223        if not self.ruta_json.exists():
224            messagebox.showwarning("Aviso", f"No existe el archivo '{self.ruta_json}' para cargar.")
225            return
226        try:
227            repo = cargar_repositorio_json(self.ruta_json, self.logger)
228            self.catalogo.objetos = repo.obtener_todos()
229            self._actualizar_tabla()
230            messagebox.showinfo(
231                "Persistencia",
232                f"Cargados {len(self.catalogo.objetos)} objetos desde '{self.ruta_json}'.",
233            )
234        except PersistenciaError as e:
235            messagebox.showerror("Error de Carga", str(e))
236
237    def _on_eliminar_objeto(self) -> None:
238        """Elimina el objeto celeste seleccionado controlando los elementos protegidos."""
239        seleccion = self.tabla.selection()
240        if not seleccion:
241            messagebox.showwarning(
242                "Aviso", "Por favor, selecciona un objeto de la tabla para eliminar."
243            )
244            return
245
246        item = self.tabla.item(seleccion)
247        valores = item.get("values")
248        if not valores:
249            return
250
251        identificador_a_borrar = valores[0]
252
253        # Lista de objetos protegidos por defecto del catálogo de ejemplo
254        PROTEGIDOS = {"Sol", "Luna", "Sirio", "Júpiter", "Saturno", "Neptuno", "M45", "Pluton"}
255
256        if identificador_a_borrar in PROTEGIDOS:
257            messagebox.showerror(
258                "Acción no permitida",
259                f"El objeto '{identificador_a_borrar}' forma parte de los cuerpos celestes base del observatorio y NO se puede eliminar.",
260            )
261            return
262
263        confirmar = messagebox.askyesno(
264            "Confirmar eliminación",
265            f"¿Estás seguro de que deseas eliminar el objeto personalizado '{identificador_a_borrar}'?",
266        )
267        if not confirmar:
268            return
269
270        # Intentamos borrar usando el método protegido del catálogo
271        try:
272            exito = self.catalogo.eliminar(identificador_a_borrar)
273            if exito:
274                self._actualizar_tabla()
275                self.logger.info(
276                    "Objeto '%s' eliminado desde la interfaz gráfica.", identificador_a_borrar
277                )
278                messagebox.showinfo(
279                    "Éxito", f"Objeto '{identificador_a_borrar}' eliminado correctamente."
280                )
281        except ValueError as e:
282            messagebox.showerror("Error de Dominio", str(e))
