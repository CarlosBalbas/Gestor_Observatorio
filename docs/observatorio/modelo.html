 50class ObjetoCeleste:
 51    """Representa un cuerpo o fenómeno astronómico observado con coordenadas ecuatoriales."""
 52
 53    def __init__(self, identificador: str, tipo: str, coordenadas: tuple[float, float]) -> None:
 54        """Inicializa un objeto celeste validando su identificador, tipo y coordenadas.
 55
 56        Args:
 57            identificador: Nombre o código único del objeto celeste.
 58            tipo: Categoría astronómica (Estrella, Planeta, Galaxia, etc.).
 59            coordenadas: Tupla (ascensión_recta, declinación) en grados.
 60
 61        Raises:
 62            TypeError: Si el identificador o el tipo no son cadenas de texto.
 63            ValueError: Si los textos están vacíos o las coordenadas están fuera de rango.
 64        """
 65        if not isinstance(identificador, str) or not isinstance(tipo, str):
 66            raise TypeError("El identificador y el tipo tienen que ser cadenas de texto")
 67        if not identificador.strip() or not tipo.strip():
 68            raise ValueError("El identificador y el tipo NO pueden estar vacíos")
 69        if len(coordenadas) != 2:
 70            raise ValueError("Se necesitan exactamente 2 coordenadas, AR y declinación")
 71        ar, dec = (float(valor) for valor in coordenadas)
 72        if ar < 0 or ar > 360 or dec < -90 or dec > 90:
 73            raise ValueError("Coordenadas fuera de rango")
 74
 75        self._identificador = identificador.strip()
 76        self._tipo = tipo.strip()
 77        self.__coordenadas = (ar, dec)
 78
 79    @property
 80    def identificador(self) -> str:
 81        """Devuelve el identificador único del objeto celeste."""
 82        return self._identificador
 83
 84    @property
 85    def tipo(self) -> str:
 86        """Devuelve el tipo astronómico del objeto celeste."""
 87        return self._tipo
 88
 89    @property
 90    def coordenadas(self) -> tuple[float, float]:
 91        """Devuelve la tupla (AR, DEC) del objeto celeste."""
 92        return self.__coordenadas
 93
 94    def a_diccionario(self) -> dict[str, object]:
 95        """Serializa la instancia a un diccionario compatible con JSON."""
 96        return {
 97            "identificador": self.identificador,
 98            "tipo": self.tipo,
 99            "coordenadas": list(self.coordenadas),
100        }
101
102    @classmethod
103    def desde_diccionario(cls, datos: dict[str, object]) -> "ObjetoCeleste":
104        """Reconstruye un ObjetoCeleste a partir de un diccionario."""
105        coords_raw = datos.get("coordenadas")
106        if not isinstance(coords_raw, (list, tuple)) or len(coords_raw) != 2:
107            raise ValueError("Formato de coordenadas inválido en diccionario")
108        coord = (float(coords_raw[0]), float(coords_raw[1]))
109        return cls(str(datos["identificador"]), str(datos["tipo"]), coord)
110
111    def __repr__(self) -> str:
112        return f"ObjetoCeleste({self.identificador!r}, {self.tipo!r}, {self.coordenadas!r})"
113
114    def __eq__(self, otro: object) -> bool | NotImplementedType:
115        if not isinstance(otro, ObjetoCeleste):
116            return NotImplemented
117        return (self.identificador, self.tipo, self.coordenadas) == (
118            otro.identificador,
119            otro.tipo,
120            otro.coordenadas,
121        )
122
123    def __hash__(self) -> int:
124        """Hash coherente con __eq__: los atributos comparados son de solo lectura."""
125        return hash((self.identificador, self.tipo, self.coordenadas))

225def _comprobar_identificadores_unicos(objetos: list[ObjetoCeleste]) -> None:
226    """Lanza ObjetoDuplicadoError si algún identificador aparece más de una vez."""
227    vistos: set[str] = set()

234class CatalogoObservatorio:
235    """Agrupa una colección de objetos celestes bajo una capacidad máxima controlada."""
236
237    def __init__(
238        self,
239        nombre: str,
240        objetos_iniciales: list[ObjetoCeleste] | None = None,
241        factor_capacidad: int = FACTOR_CAPACIDAD_POR_DEFECTO,
242    ) -> None:
243        """Crea el catálogo y fija su capacidad máxima.
244
245        Args:
246            nombre: Nombre del catálogo.
247            objetos_iniciales: Objetos con los que arranca (se copia la lista).
248            factor_capacidad: La capacidad máxima es nº de objetos iniciales x factor
249                (con 8 como base si el catálogo empieza vacío).
250
251        Raises:
252            ValueError: Si el nombre está vacío o el factor no es un entero positivo.
253            ObjetoDuplicadoError: Si los objetos iniciales repiten identificador.
254        """
255        if not nombre.strip():
256            raise ValueError("El nombre del catálogo no puede estar vacío")
257        if type(factor_capacidad) is not int or factor_capacidad < 1:
258            raise ValueError("El factor de capacidad tiene que ser un entero positivo")
259        self.nombre = nombre.strip()
260        self.objetos: list[ObjetoCeleste] = (
261            [] if objetos_iniciales is None else list(objetos_iniciales)
262        )
263        _comprobar_identificadores_unicos(self.objetos)
264        base = len(self.objetos) if len(self.objetos) > 0 else 8
265        self.__capacidad_maxima = base * factor_capacidad
266
267    @property
268    def capacidad_maxima(self) -> int:
269        """Capacidad máxima permitida en el catálogo."""
270        return self.__capacidad_maxima
271
272    def anadir(self, objeto: ObjetoCeleste) -> None:
273        """Añade un objeto celeste verificando capacidad e identificador único.
274
275        Raises:
276            CapacidadMaximaError: Si el catálogo está lleno.
277            ObjetoDuplicadoError: Si ya existe un objeto con el mismo identificador.
278        """
279        if len(self.objetos) >= self.capacidad_maxima:
280            raise CapacidadMaximaError(
281                f"No se puede registrar '{objeto.identificador}'. Capacidad máxima alcanzada.",
282                len(self.objetos),
283                self.capacidad_maxima,
284            )
285        if self.buscar_por_id(objeto.identificador) is not None:
286            raise ObjetoDuplicadoError(
287                f"Ya existe un objeto con identificador '{objeto.identificador}'."
288            )
289        self.objetos.append(objeto)
290
291    def reemplazar_objetos(self, nuevos: list[ObjetoCeleste]) -> None:
292        """Sustituye el contenido del catálogo pasando cada objeto por anadir().
293
294        Comprueba la capacidad antes de tocar nada, de modo que si los datos no caben
295        el catálogo queda intacto. Al usar anadir(), las subclases (por ejemplo
296        CatalogoObservable) siguen notificando cada alta.
297
298        Raises:
299            CapacidadMaximaError: Si los nuevos objetos superan la capacidad máxima.
300            ObjetoDuplicadoError: Si hay identificadores repetidos en los nuevos objetos.
301        """
302        nuevos = list(nuevos)
303        _comprobar_identificadores_unicos(nuevos)
304        if len(nuevos) > self.capacidad_maxima:
305            raise CapacidadMaximaError(
306                f"No se pueden cargar {len(nuevos)} objetos. Capacidad máxima: "
307                f"{self.capacidad_maxima}.",
308                len(nuevos),
309                self.capacidad_maxima,
310            )
311        self.objetos = []
312        for objeto in nuevos:
313            self.anadir(objeto)
314
315    def buscar_por_id(self, identificador: str) -> ObjetoCeleste | None:
316        """Busca un objeto celeste por su identificador exacto."""
317        for obj in self.objetos:
318            if obj.identificador == identificador:
319                return obj
320        return None
321
322    def resumen(self) -> str:
323        """Genera un resumen del estado del catálogo (cumple el protocolo Resumible)."""
324        return (
325            f"Catálogo '{self.nombre}' con {len(self.objetos)}/{self.capacidad_maxima} "
326            f"objetos registrados"
327        )
328
329    def eliminar(self, identificador: str) -> bool:
330        """Elimina un objeto del catálogo buscando por su identificador.
331
332        Args:
333            identificador: Nombre o código único del objeto a eliminar.
334
335        Returns:
336            True si se ha eliminado con éxito, False si no existía.
337
338        Raises:
339            ValueError: Si el objeto pertenece a OBJETOS_PROTEGIDOS.
340        """
341        ident_limpio = identificador.strip()
342        if ident_limpio in OBJETOS_PROTEGIDOS:
343            raise ValueError(
344                f"No se puede eliminar '{ident_limpio}' por ser un objeto celeste "
345                "de referencia protegido."
346            )
347
348        for obj in self.objetos:
349            if obj.identificador == ident_limpio:
350                self.objetos.remove(obj)
351                return True
352        return False
